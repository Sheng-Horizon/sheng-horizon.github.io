1. $exgcd$ 把 ```a%b``` 打成了 ```a^b``` 。
   ```cpp
   LL exgcd(LL a,LL b,LL &x,LL &y){
       if(b==0){
           x=1,y=0;
           return a;
       }
       LL d=exgcd(b,a%b,x,y);
     //LL d=exgcd(b,a^b,x,y);错误
       LL z=x;
       x=y,y=z-(a/b)*y;
       return d;
   }
   ```

2. 取模模多了

   * 在[P5170 【模板】类欧几里得算法](https://www.luogu.com.cn/problem/P5170)中

   ```cpp
   pack calc(LL a,LL b,LL c,LL n){
       LL m=(a*n+b)/c;
     //LL m=(a*n+b)/c%MOD;错误
     
       LL normSum=n+1;//和
       LL lineSum=n*(n+1)%MOD*inv2%MOD;//单调队列和
       LL squrSum=n*(n+1)%MOD*(2*n+1)%MOD*inv6%MOD;//平方队列和
       LL AC=a/c,BC=b/c;
       pack res;
   
       if(a==0){//递归必须有边界！
           res.f=normSum*BC%MOD;
           res.g=lineSum*BC%MOD;
           res.h=BC*BC%MOD*normSum%MOD;
           return res;
       }
       if(a>=c||b>=c){
           pack lastGen=calc(a%c,b%c,c,n);
           res.f=(lineSum*AC%MOD+normSum*BC%MOD+lastGen.f)%MOD;
           res.g=(squrSum*AC%MOD+lineSum*BC%MOD+lastGen.g)%MOD;
           res.h=(squrSum*AC%MOD*AC%MOD+normSum*BC%MOD*BC%MOD+lineSum*2%MOD*AC%MOD*BC%MOD+2*AC*lastGen.g%MOD+2*BC*lastGen.f%MOD+lastGen.h)%MOD;
           return res;
       }else{
           pack fGen=calc(c,c-b-1,a,m-1);
           res.f=(m*n%MOD-fGen.f+MOD)%MOD;
           res.g=((m*n%MOD*(n+1)%MOD*inv2%MOD)-inv2*((fGen.h+fGen.f)%MOD)%MOD+MOD)%MOD;
           res.h=((m*n%MOD*(m+1)%MOD)-2*((fGen.g+fGen.f)%MOD)%MOD-res.f+MOD)%MOD;
           return res;
       }
   }
   ```

5. 高斯消元 `raxio` 除反了

  * 在[Broken robot](https://www.luogu.com.cn/problem/CF24D)中

  ```cpp
  #include<cstring>
  #include<cstdio>
  #include<iostream>
  using namespace std;
  const int N=1010;
  int n,m,x,y;
  double f[N][N];
  double fac[N][N];
  
  int main(){
      // freopen("test.out","w",stdout);
      freopen("test.in","r",stdin);
      scanf("%d%d%d%d",&n,&m,&x,&y);
      if(m==1){
          printf("%d\n",2*(n-x));
          return 0;
      }
      for(int i=n-1;i>=1;i--){//处理第i行
          //初始化高斯消元矩阵
          for(int j=1;j<m;j++) fac[j+1][j]=fac[j][j+1]=-1;
          for(int j=2;j<m;j++) fac[j][j]=3;
          fac[1][1]=fac[m][m]=2;//经过演算得到的构造
          for(int j=2;j<m;j++) fac[j][m+1]= 4+f[i+1][j];
          fac[1][m+1]=3+f[i+1][1];
          fac[m][m+1]=3+f[i+1][m];
          //开始消元（不安套路出牌）
          //1. 从上到下消一遍
          for(int j=1;j<m;j++){
              double raxio=fac[j+1][j]/fac[j][j];
              fac[j+1][j]=0;//直接处决（不是
              fac[j+1][j+1]-=fac[j][j+1]*raxio;
              fac[j+1][m+1]-=fac[j][m+1]*raxio;
          }
          //目前状态，除了第n排之外，都留了了一个小尾巴
        
        
          for(int j=m;j>1;j--){
              double raxio=fac[j-1][j]/fac[j][j];
            //double raxio=fac[j][j]/fac[j-1][j];
            //错误的！！！！
              fac[j-1][j]=0;
              fac[j-1][m+1]-=fac[j][m+1]*raxio;
          }
        
        
        
          // back 1
          for(int j=1;j<=m;j++) fac[j][m+1]/=fac[j][j],fac[j][j]=1;
          //干净了
          for(int j=1;j<=m;j++) f[i][j]=fac[j][m+1];
      }
      printf("%lf\n",f[x][y]);
      return 0;
  }
  ```
