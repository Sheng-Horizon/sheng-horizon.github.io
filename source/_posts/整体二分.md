整体二分太玄学了，半天没看明白。

# 啥时候该用整体二分？

假设你现在有多个询问 **在同一范围内** 的二分，如果一个一个二分的话，就会有很多地方重复访问，造成复杂度不优。

这个时候你就可以用整体二分：不管是修改还是询问都一股脑儿放一起，这样对于多个询问，修改只需要做一次就行了。

当然前提是修改没有 **”前效性“**——某次修改对前面无影响。

# 复杂度怎么分析呢？

按照我的理解，先假设没有修改，全是询问。

* 通常情况下有 $O(n)$ 个询问。
* 然后呢，每个询问最多会因为二分被处理 $O(\log n)$ 次。
* 然后呢，又因为每次询问——通常是在梳妆数组上，的复杂度是 $O(\log n)$ 的。
* 所以，只看询问的总复杂度通常是 $O(n \log^2 n)$ 的。

接下来，带上修改

* 其实现在就很好理解了，你可以想成修改和询问一样，都是一次操作，会被二分处理 $O(\log n)$ 次，每次在树状数组上处理复杂度 $O(\log n)$，然后有 $O(n)$ 个修改，于是复杂度也是 $O(n \log^2 n)$。

于是总复杂度就是 $O(n \log^2 n)$。

# 示例代码

* [P3834 【模板】可持久化线段树 2](https://www.luogu.com.cn/problem/P3834)

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+10;
int a[N],n,m,tot,INF,ans[N],mp[N],p[N];
vector<int> lsh;

struct node{
    int op,l,r,k,id;
    //op==2 查询: l,r,k,id;
    //op==1 修改: size,val,null,id
}cache[2*N],CL[2*N],CR[2*N];

int lowbit(int x){return x&(-x);}
void add(int x,int v){
    while(x<=n){
        p[x]+=v;
        x+=lowbit(x);
    }
}

int query(int x){
    int v=0;
    while(x){
        v+=p[x];
        x-=lowbit(x);
    }
    return v;
}

void solve(int l,int r,int Ql,int Qr){//值左右，操作区间左右
    if(Ql>Qr) return; //已经没有可以操作的了，没有必要继续了
    if(l==r){
        for(int i=Ql;i<=Qr;i++) if(cache[i].op==2) ans[cache[i].id]=l;
        //已经只剩一个选项了，不用再修改判断，直接记录答案
        return;
    }
    int mid=l+(r-l)/2;
    int totL=0,totR=0;
    for(int i=Ql;i<=Qr;i++){
        if(cache[i].op==1){
            if(cache[i].l<=mid) CL[++totL]=cache[i],add(cache[i].id,cache[i].r);
            else CR[++totR]=cache[i];
        }else{
            int tmp=query(cache[i].r)-query(cache[i].l-1);
            if(cache[i].k<=tmp) CL[++totL]=cache[i];
            else cache[i].k-=tmp,CR[++totR]=cache[i];
        }
    }
    //复原
    for(int i=1;i<=totL;i++) if(CL[i].op==1) add(CL[i].id,-CL[i].r);
    for(int i=1;i<=totL;i++) cache[Ql+i-1]=CL[i];
    for(int i=1;i<=totR;i++) cache[Ql+i+totL-1]=CR[i];
    solve(l,mid,Ql,Ql+totL-1);//线段树般的递归
    solve(mid+1,r,Ql+totL,Qr);
}

int main(){
    // freopen("test.in","r",stdin);
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++){
        scanf("%d",&a[i]);
        lsh.push_back(a[i]);
    }
    sort(lsh.begin(),lsh.end());
    lsh.erase(unique(lsh.begin(),lsh.end()),lsh.end());
    INF=lsh.size()+1;
    for(int i=1;i<=n;i++){
        int tmp=lower_bound(lsh.begin(),lsh.end(),a[i])-lsh.begin()+1;
        mp[tmp]=a[i];
        a[i]=tmp;
        cache[++tot]={1,a[i],1,0,i};
    }
    for(int i=1;i<=m;i++){
        int l,r,k;
        scanf("%d%d%d",&l,&r,&k);
        cache[++tot]={2,l,r,k,i};
    }
    solve(0,INF,1,tot);
    for(int i=1;i<=m;i++) printf("%d\n",mp[ans[i]]); 
    return 0;
}
```

前面讲到，整体二分不管是查询还是修改都只当是一个“操作”来处理。因此整体二分自带中途修改的能力。

于是乎，做 [P2617 Dynamic Rankings](https://www.luogu.com.cn/problem/P2617) 连整体二分部分都不用改，只改改 `main` 函数就行了。

```cpp
int main(){
    // freopen("test.in","r",stdin);
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++){
        scanf("%d",&a[i]);
        cache[++tot]={1,a[i],1,0,i};
    }
    for(int i=1;i<=m;i++){
        char op;
        int l,r,k,x,y;
        scanf("\n%c ",&op);
        // printf("%c\n",op);
        if(op=='Q'){
            scanf("%d%d%d",&l,&r,&k);
            cache[++tot]={2,l,r,k,++cnt};
        }else{
            scanf("%d%d",&x,&y);
            cache[++tot]={1,a[x],-1,0,x};
            cache[++tot]={1,y,1,0,x};
            a[x]=y;
        }
        
    }
    solve(-INF,INF,1,tot);
    for(int i=1;i<=cnt;i++) printf("%d\n",ans[i]); 
    return 0;
}
```
