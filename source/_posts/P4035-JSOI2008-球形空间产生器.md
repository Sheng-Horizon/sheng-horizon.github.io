## 高斯消元

```cpp
#include<iostream>
#include<cstdio>
#include<algorithm>
#include<cstring>
#include<cmath>
using namespace std;
const int N=20;
double a[N][N];
double fac[N][N];
int n;
double last;
double eps=1e-6;

void guess(){
    int cnt=0,t=0;
    double rate=0;
    for(int i=1;i<=n;i++){
        t=0;
        for(int j=cnt+1;j<=n;j++) if(fabs(fac[j][i])-fabs(fac[t][i])>eps) t=j;
        swap(fac[++cnt],fac[t]);
        for(int j=1;j<=n;j++){
            if(j==cnt) continue;
            rate=fac[j][i]/fac[cnt][i];
            for(int k=i;k<=n+1;k++){
                fac[j][k]-=rate*fac[cnt][k];
            }
        }
    }
    // return 0;
}

int main(){
    freopen("test.in","r",stdin);
    scanf("%d",&n);
    for(int i=1;i<=n+1;i++) for(int j=1;j<=n;j++) scanf("%lf",&a[i][j]);
    for(int i=1;i<=n;i++){
        last=0;
        for(int j=1;j<=n;j++){
            fac[i][j]=2*(a[i][j]-a[i+1][j]);
            last+=(a[i][j]*a[i][j]);
            last-=(a[i+1][j]*a[i+1][j]);
        }
        fac[i][n+1]=last;
    }
    guess();
    for(int i=1;i<=n;i++){
        printf("%.3lf ",fac[i][n+1]/fac[i][i]);
    }
    return 0;
}
```

## 模拟退火

```cpp
#include<iostream>
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<cmath>
using namespace std;
int n;
// double average[15];
struct node{
    double x[15];
}a[15],ans;
double minVar=1e18;
double esp=1e-7;

double calc(node x){
    double dist[15];
    double total=0;
    for(int i=1;i<=n+1;i++){
        dist[i]=0;
        for(int j=1;j<=n;j++){
            double dx=a[i].x[j]-x.x[j];
            dist[i]+=dx*dx;
        }
        total+=dist[i];
    }
    total/=(n+1);
    double res=0;
    for(int i=1;i<=n+1;i++) res+=(dist[i]-total)*(dist[i]-total);
    // res/=(n+1);
    if(res-minVar<esp){
        minVar=res;
        ans=x;
    }
    return res;
}

double rand(double l,double r){
    return (double) rand()/RAND_MAX*(r-l)+l;
}

void SA(){
    node cur=ans;
    // for(int i=1;i<=n;i++) cur.x[i]=rand(-20000,20000);
    for(double t=8e3;t>=1e-6;t=t*0.999){
        node newPoint;
        for(int i=1;i<=n;i++) newPoint.x[i]=rand(cur.x[i]-t,cur.x[i]+t);
        double dealta=calc(newPoint)-calc(cur);//负数：最好，正数：概率
        if(exp(-dealta/t)*RAND_MAX>rand()) cur=newPoint;
    }
}

int main(){
    freopen("test.in","r",stdin);
    srand(time(NULL));
    scanf("%d",&n);
    for(int i=1;i<=n+1;i++){
        for(int j=1;j<=n;j++){
            scanf("%lf",&a[i].x[j]);
            ans.x[j]+=a[i].x[j];
        }
    }
    for(int i=1;i<=n;i++) ans.x[i]/=(n+1);
    minVar=calc(ans);
    while((double)clock()/CLOCKS_PER_SEC<0.9) SA();
    for(int i=1;i<=n;i++) printf("%.3lf ",ans.x[i]);
    puts("");
    return 0;
}
```
