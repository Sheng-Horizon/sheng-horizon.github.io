用各种奇奇怪怪的线段树处理“某某第 $k$ 小”的东西我是一直都不太明白，写一篇记录一下吧。

# 静态全局第 $k$ 小
最简单的方法是排一次序。

但是如果我要拿线段树做呢？

可以选择开一颗 **权值线段树**。假设某次要查询第 $k$ 小，就看左子树中的值的数量是否 $\ge k$，若是，则递归求左子树第 $k$ 小；否则，设左子树中值得数量为 $cnt$，则求右子树第 $cnt-k$ 小。

# 静态区间第 $k$ 小

还是考虑使用权值线段树。

但是怎么做捏？

其实可以考虑对权值线段树进行前缀和。即对每个位置都建一个权值线段树，其中的内容=前一棵权值线段树+现在这个位置的值

假设目前要求区间 $[l,r]$ 的第 $k$ 小，此时递归求左子树的 $cnt$ 时就不是单纯的 $cnt$ 了，而是第 $r$ 棵线段树的 $cnt_r$ 减去第 $l-1$ 棵线段树的 $cnt_{l-1}$。

很明显直接硬开 $n$ 棵线段树那是妥妥的浪费啊。于是又发现，每次加入一个新值实际上最多只需要新加 $O(\log n)$ 个节点，于是就可以用 **可持久化权值线段树** 力。

``` cpp
#include<bits/stdc++.h>
using namespace std;
const int N=2e5+10;
const int M=40*N;
int a[N];
int mp[N];
int tot;
int root[N];

vector<int> lsh;
struct node{
    int l,r,val;
}tr[M];
int n,m;

void pushup(int k){
    tr[k].val=tr[tr[k].l].val+tr[tr[k].r].val;
}

void update(int &k,int prev,int l,int r,int x,int v){
    k=++tot;
    //因为要修改，就直接tot++了，不要和可持久化平衡树搞混了
    tr[k]=tr[prev];
    if(l==r){
        tr[k].val+=v;
        return;
    }
    int mid=l+(r-l)/2;
    if(x<=mid) update(tr[k].l,tr[prev].l,l,mid,x,v);
    else update(tr[k].r,tr[prev].r,mid+1,r,x,v);
    pushup(k);
}

int query(int kR,int kL,int l,int r,int k){
    if(l==r) return mp[l];
    int cnt=tr[tr[kR].l].val-tr[tr[kL].l].val;
    int mid=l+(r-l)/2;
    if(cnt>=k) return query(tr[kR].l,tr[kL].l,l,mid,k);
    else return query(tr[kR].r,tr[kL].r,mid+1,r,k-cnt);
}

int main(){
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++){
        scanf("%d",&a[i]);
        lsh.push_back(a[i]);
    }
    sort(lsh.begin(),lsh.end());
    lsh.erase(unique(lsh.begin(),lsh.end()),lsh.end());
    for(int i=1;i<=n;i++){
        int t=lower_bound(lsh.begin(),lsh.end(),a[i])-lsh.begin()+1;
        mp[t]=a[i];
        a[i]=t;
    }
    int total=lsh.size();
    for(int i=1;i<=n;i++){
        update(root[i],root[i-1],1,total,a[i],1);
    }
    for(int i=1;i<=m;i++){
        int l,r,k;
        scanf("%d%d%d",&l,&r,&k);
        printf("%d\n",query(root[r],root[l-1],1,total,k));   
    }
    return 0;
}
```

# 动态区间第 $k$ 小
动态区间第 $k$ 小是我最不理解的。

这里的原理和 前缀和与树状数组 的关系是一样的。
如果前缀和需要修改的话，其复杂度是 $O(n)$ 的，于是便引入 树状数组或线段树，牺牲查询的复杂度，而把修改的复杂度降下来。

我们可以把处理 静态区间第 $k$ 小 的 可持久化权值线段树 当成一个第一位是前缀和，第二维是线段树的一个数据结构，此时我们为了处理修改，就把第一维换成 树状数组 或者 线段树。

于是，动态区间第 $k$ 小 就可以拿树套树做了。

* 树状数组套线段树

``` cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 5e4+100;
const int INF = 1e8+100;
vector<int> cacher, cachel;
int n, m;
int a[N];
int root[N];
int tot;
struct seg
{
    int lson[N * 400], rson[N * 400], val[N * 400];
    void update(int &k, int l, int r, int x, int v)
    {
        if (!k) k = ++tot;
        if (l == r){
            val[k] += v;
            return;
        }
        int mid = l + (r - l) / 2;
        if (x <= mid) update(lson[k], l, mid, x, v);
        else update(rson[k], mid + 1, r, x, v);
        val[k] = val[lson[k]] + val[rson[k]];
    }
    int getRank(int l, int r, int x)
    {
        int sum=0;
        if(l==r){
            for(int i:cacher) sum+=val[i];
            for(int i:cachel) sum-=val[i];
            return sum;
        }
        int mid=l+(r-l)/2;
        if(x<=mid){
            for(int i=0;i<cacher.size();i++) cacher[i]=lson[cacher[i]];
            for(int i=0;i<cachel.size();i++) cachel[i]=lson[cachel[i]];
            return getRank(l,mid,x);
        }else{
            for(int i:cacher) sum+=val[lson[i]];
            for(int i:cachel) sum-=val[lson[i]];
            for(int i=0;i<cacher.size();i++) cacher[i]=rson[cacher[i]];
            for(int i=0;i<cachel.size();i++) cachel[i]=rson[cachel[i]];
            return sum+getRank(mid+1,r,x);
        }
    }
    int getVal(int l,int r,int k){//知道排名求值
        if(l==r) return l;
        int sum=0;
        for(int i:cacher) sum+=val[lson[i]];
        for(int i:cachel) sum-=val[lson[i]];
        int mid=l+(r-l)/2;
        if(sum<k){
            for(int i=0;i<cacher.size();i++) cacher[i]=rson[cacher[i]];
            for(int i=0;i<cachel.size();i++) cachel[i]=rson[cachel[i]];
            return getVal(mid+1,r,k-sum);
        }else{
            for(int i=0;i<cacher.size();i++) cacher[i]=lson[cacher[i]];
            for(int i=0;i<cachel.size();i++) cachel[i]=lson[cachel[i]];
            return getVal(l,mid,k);
        }
    }

} tr;
int lowbit(int x) {return x & (-x);}

void update(int x, int p, int v) {for (int i = x; i <= n; i += lowbit(i)) tr.update(root[i], 0, INF, p, v);}

int getRank(int l, int r, int k){
    cachel.clear();
    cacher.clear();
    for (int i = r; i; i -= lowbit(i)) cacher.push_back(root[i]);
    for (int i = l - 1; i; i -= lowbit(i)) cachel.push_back(root[i]);
    return tr.getRank(0, INF, k);
}

int getVal(int l,int r,int k){
    cachel.clear();
    cacher.clear();
    for (int i = r; i; i -= lowbit(i)) cacher.push_back(root[i]);
    for (int i = l - 1; i; i -= lowbit(i)) cachel.push_back(root[i]);
    return tr.getVal(0,INF,k);
}

int main()
{
    freopen("test.in","r",stdin);
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i++){
        scanf("%d", &a[i]);
        update(i, a[i], 1);
    }
    for (int i = 1; i <= m; i++){
        int op, l, r, k, pos;
        scanf("%d", &op);
        if (op == 1){
            scanf("%d%d%d", &l, &r, &k);
            printf("%d\n", getRank(l, r, k - 1) + 1);
        }
        else if (op == 2){
            scanf("%d%d%d", &l, &r, &k);
            printf("%d\n", getVal(l, r, k));
        }else if(op==3){
            scanf("%d%d",&pos,&r);
            update(pos,a[pos],-1);
            update(pos,r,1);
            a[pos]=r;
        }else if(op==4){
            scanf("%d%d%d", &l, &r, &k);
            int rank=getRank(l,r,k-1);
            if(rank<1) puts("-2147483647");
			else printf("%d\n",getVal(l,r,rank));
        }else{
            scanf("%d%d%d",&l,&r,&k);
			int kk=getRank(l,r,k);
			if(kk>=(r-l+1)) puts("2147483647");
			else printf("%d\n",getVal(l,r,kk+1));
        }
    }
    return 0;
}
```

* 线段树套线段树版本

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=4e7;
const int M=5e4+10;
const int K=15000000;
vector<int> cache;
int a[M];
const int INF=1e8+10;
int root,tot;
int lson[K],rson[K];

struct seg{
    int lson[N],rson[N],val[N];//权值线段树
    int root[N];
    int tot=0;
    void pushup(int k){
        val[k]=val[lson[k]]+val[rson[k]];
    }
    void update(int &k,int l,int r,int x,int v){
        if(!k) k=++tot;
        if(l==r){
            val[k]+=v;
            return;
        }
        int mid=l+(r-l)/2;
        if(x<=mid) update(lson[k],l,mid,x,v);
        else update(rson[k],mid+1,r,x,v);
        pushup(k);
    }
    int getRank(int l,int r,int x){
        int res=0;
        if(l==r){
            for(int k:cache) res+=val[k];
            return res;
        }
        int mid=l+(r-l)/2;
        if(x<=mid){
            for(int i=0;i<cache.size();i++) cache[i]=lson[cache[i]];
            return getRank(l,mid,x);
        }else{
            for(int k:cache) res+=val[lson[k]];
            for(int i=0;i<cache.size();i++) cache[i]=rson[cache[i]];
            return res+getRank(mid+1,r,x);
        }
    }
    int getVal(int l,int r,int x){
        if(l==r) return l;
        int mid=l+(r-l)/2;
        int res=0;
        for(int k:cache) res+=val[lson[k]];
        if(x<=res){
            for(int i=0;i<cache.size();i++) cache[i]=lson[cache[i]];
            return getVal(l,mid,x);
        }else{
            for(int i=0;i<cache.size();i++) cache[i]=rson[cache[i]];
            return getVal(mid+1,r,x-res);
        }
    }
}tr;

void get(int &k,int l,int r,int x,int y){
    if(!k) k=++tot;
    if(x<=l&&r<=y){
        cache.push_back(tr.root[k]);
        return;
    }
    int mid=l+(r-l)/2;
    if(x<=mid) get(lson[k],l,mid,x,y);
    if(mid<y) get(rson[k],mid+1,r,x,y); 
}

void update(int &k,int l,int r,int x,int v,int va){
    if(!k) k=++tot;
    tr.update(tr.root[k],0,INF,v,va);
    if(l==r) return;
    int mid=l+(r-l)/2;
    if(x<=mid) update(lson[k],l,mid,x,v,va);
    else update(rson[k],mid+1,r,x,v,va);
}

int main(){
    freopen("test.in","r",stdin);
    freopen("test.out","w",stdout);
    int n,m;
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++){
        cache.clear();
        scanf("%d",&a[i]);
        update(root,1,n,i,a[i],1);
    }
    for(int i=1;i<=m;i++){
        int op,x,y,z;
        scanf("%d",&op);
        if(op==3) scanf("%d%d",&x,&y);
        else scanf("%d%d%d",&x,&y,&z);
        if(op==1){
            cache.clear();
            get(root,1,n,x,y);
            printf("%d\n",tr.getRank(0,INF,z-1)+1);
        }else if(op==2){
            cache.clear();
            get(root,1,n,x,y);
            printf("%d\n",tr.getVal(0,INF,z));
        }else if(op==3){
            update(root,1,n,x,a[x],-1);
            update(root,1,n,x,y,1);
            a[x]=y;
        }else if(op==4){
            cache.clear();
            get(root,1,n,x,y);
            int k=tr.getRank(0,INF,z-1);
            if(k){
                cache.clear();
                get(root,1,n,x,y);
                printf("%d\n",tr.getVal(0,INF,k));
            }else{
                puts("-2147483647");
            }
        }else if(op==5){
            cache.clear();
            get(root,1,n,x,y);
            int k=tr.getRank(0,INF,z);
            if(k>=(y-x+1)){
                puts("2147483647");
            }else{
                cache.clear();
                get(root,1,n,x,y);
                printf("%d\n",tr.getVal(0,INF,k+1));
            }
        }
    }
    return 0;
}
```
