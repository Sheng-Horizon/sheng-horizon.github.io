# Tons of Formulas and Theorems

1. 欧拉公式（同时也是复数指数形式的定义）：$e^{ix}=\cos x+i\sin x$
2. 反三角函数：$acos(-1)=\pi$
3. 单位复数根：$\omega_{n}^{k}=e^{2\pi i k/n}=\cos\ k *\frac{2\pi}{n}+i\sin k*\frac{2\pi}{n}$ 
4. 消去定理：对于任何整数 $n\ge 0,k\ge 0$，以及 $d>0$，

$$
\omega_{dn}^{dk}=\omega_n^k
$$


​		推论：对于任意偶数 $n>0$，有 $\omega_n^{n/2}=\omega_2=-1$

5. 折半定理：

   > 如果 $n>0$ 为偶数，那么 $n$ 个 $n$ 次单位复数根的平方的集合就是 $n/2$ 个 $n/2$ 次单位复数根的集合。
   >
   > $(\omega_n^{k+n/2})^2=\omega_n^{2k+n}=\omega_n^{2k}\omega^n_n=\omega_n^{2k}=(\omega_n^k)^2$
   > $$
   > \begin{align}
   > y_k &= y_k^{[0]}+\omega_n^k y^{[1]}_k\\
   > 		&= A^{[0]}(\omega_n^{2k})+\omega^k_n A^{[1]}(\omega_n^{2k})\\
   > 		&= A(\omega^k_n)
   > \end{align}
   > $$
   >
   > $$
   > \begin{align}
   > y_{k+(n/2)} &= y_k^{[0]}-\omega_n^k y^{[1]}_k\\
   > 		&= y_k^{[0]}-\omega_n^{k+(n/2)} y^{[1]}_k\\
   > 		&= A^{[0]}(\omega_n^{2k})+\omega^{k+(n/2)}_n A^{[1]}(\omega_n^{2k})\\
   > 		&= A^{[0]}(\omega_n^{2k+n})+\omega^{k+(n/2)}_n A^{[1]}(\omega_n^{2k+n})\\
   > 		&= A(\omega^{k+(n/2)}_n)
   > \end{align}
   > $$

```cpp
#include<iostream>
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<cmath>
using namespace std;
const int N=3e6+10;
int a,b;
const double pi=acos(-1);
struct complex{
    double x=0,y=0;//x+yi;
    complex(double a=0,double b=0){x=a,y=b;}
    complex operator + (const complex &b) const{return complex(x+b.x,y+b.y);}
    complex operator - (const complex &b) const{return complex(x-b.x,y-b.y);}
    complex operator * (const complex &b) const{return complex(x*b.x-y*b.y,x*b.y+y*b.x);} 
}f[N],g[N];
int rev[N];

void change(complex v[],int len){
    for(int i=0;i<len;i++){
        rev[i]=rev[i>>1]>>1;
        if(i&1) rev[i]|=len>>1;
    }
    for(int i=0;i<len;i++) if(i<rev[i]) swap(v[i],v[rev[i]]);
    return;
}

void FFT(complex v[],int len,int type){
    change(v,len);
    for(int n=2;n<=len;n<<=1){
        complex wn(cos(2*pi/n),sin(type*2*pi/n));
        for(int j=0;j<len;j+=n){
            complex w(1,0);
            for(int k=j;k<j+n/2;k++){
                complex u=v[k];
                complex t=w*v[k+n/2];
                v[k]=u+t;
                v[k+n/2]=u-t;
                w=w*wn;
            }
        }
    }
    if(type==-1){
        for(int i=0;i<len;i++) v[i].x/=len;
    }
}

int main(){
    freopen("test.in","r",stdin);
    scanf("%d%d",&a,&b);
    for(int i=0;i<=a;i++) scanf("%lf",&f[i].x);
    for(int i=0;i<=b;i++) scanf("%lf",&g[i].x);//复数乘复数，好使
    int mi=ceil(log2(a+b+1)),length=1<<mi;
    FFT(f,length,1);//DFT;
    FFT(g,length,1);
    for(int i=0;i<length;i++) f[i]=f[i]*g[i];//转点权
    FFT(f,length,-1);//IDFT;
    for(int i=0;i<=a+b;i++) printf("%d ",(int)(f[i].x+0.5));
    return 0;
}
```
