线段树有一个特点，就是不管是怎样的一颗，其结构都是一致的。
所以相比 `FHQ-Treap` 的合并与分裂而言，要相对简单一些。

关键在于合并的两个线段树有多少个叶子结点重合，假设有 $m$ 个叶子结点重合，每个结点 $O(\log n)$ 次合并。所以总复杂度 $O(m \log n)$，如果直接当线段树合并是 $O(\log n)$ 的话，就等着寄吧。（

线段树合并是基于动态开点的。

没了。

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=100100;
const int M=18*18*N;
int n,m;
int h[N],ne[2*N],e[2*N],idx;
int f[N][20];
int depth[N];
int ans[N];

struct seg{
    int val[M],lson[M],rson[M];
    int tot=0;
    int root[M];

    void update(int &k,int l,int r,int x,int v){
        if(!k) k=++tot;
        if(l==r){
            val[k]+=v;
            return;
        }
        int mid=l+(r-l)/2;
        if(x<=mid) update(lson[k],l,mid,x,v);
        else update(rson[k],mid+1,r,x,v);
        val[k]=max(val[lson[k]],val[rson[k]]);
    }

    int query(int k,int l,int r){
        if(!k||!val[k]) return 0;
        if(l==r) return r;
        int mid=l+(r-l)/2;
        if(val[lson[k]]>=val[rson[k]]) return query(lson[k],l,mid);
        else return query(rson[k],mid+1,r);
    }

    int merge(int k1,int k2,int l,int r){
        if(!k1||!k2) return k1+k2;
        if(l==r){
            val[k1]+=val[k2];
            return k1;
        }
        int mid=l+(r-l)/2;
        lson[k1]=merge(lson[k1],lson[k2],l,mid);
        rson[k1]=merge(rson[k1],rson[k2],mid+1,r);
        val[k1]=max(val[lson[k1]],val[rson[k1]]);
        return k1;
    }

    void debug(int k,int l,int r,int layer){
        for(int i=1;i<=layer;i++) printf("  ");
        if(k==0){
            puts("|__BLANK");
            return;
        } 
        printf("|__%d %d %d\n",l,r,val[k]);
        if(l==r) return;
        int mid=l+(r-l)/2;
        debug(lson[k],l,mid,layer+1);
        debug(rson[k],mid+1,r,layer+1);
    }
    
}tr;

void add(int a,int b){
    e[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}


void dfs(int point,int father){
    depth[point]=depth[father]+1;
    f[point][0]=father;
    for(int i=1;i<20;i++) f[point][i]=f[f[point][i-1]][i-1];
    for(int i=h[point];i!=-1;i=ne[i]){
        int j=e[i];
        if(j==father) continue;
        dfs(j,point);
    }
}

void getAns(int point,int father){
    for(int i=h[point];i!=-1;i=ne[i]){
        int j=e[i];
        if(j==father) continue;
        getAns(j,point);
        tr.root[point]=tr.merge(tr.root[point],tr.root[j],1,N);
    }
    
    ans[point]=tr.query(tr.root[point],1,N);
}

int lca(int a,int b){
    if(depth[a]<depth[b]) swap(a,b);
    for(int i=19;i>=0;i--){
        if(depth[f[a][i]]>=depth[b]) a=f[a][i];
    }
    if(a==b) return a;
    for(int i=19;i>=0;i--){
        if(f[a][i]!=f[b][i]) a=f[a][i],b=f[b][i];
    }
    return f[a][0];
}

int main(){
    // freopen("test.in","r",stdin);
    memset(h,-1,sizeof h);
    scanf("%d%d",&n,&m);
    for(int i=1;i<n;i++){
        int a,b;
        scanf("%d%d",&a,&b);
        add(a,b);
        add(b,a);
    }
    dfs(1,0);
    for(int i=1;i<=m;i++){
        int a,b,c;
        scanf("%d%d%d",&a,&b,&c);
        int LCA=lca(a,b);
        // printf("%d %d %d\n",a,b,LCA);
        tr.update(tr.root[a],1,N,c,1);
        tr.update(tr.root[b],1,N,c,1);
        tr.update(tr.root[LCA],1,N,c,-1);
        tr.update(tr.root[f[LCA][0]],1,N,c,-1);
    }
    getAns(1,0);
    for(int i=1;i<=n;i++) printf("%d\n",ans[i]);
    return 0;
}
```

# [P5494 【模板】线段树分裂](https://www.luogu.com.cn/problem/P5494)

由于线段树结构一致，遇到要分裂的结点直接复制到新树上，然后把旧树上的结点删了就行了。

```cpp
#include<iostream>
#include<cstdio>
#include<cstring>
using namespace std;
const int N=2e5+10;
const int M=40*N;
int n,m;
int root[N],TC;//TreeCnt

struct node{
    int lson[M],rson[M];
    long long val[M];
    int tot;

    void pushup(int k){
        val[k]=val[lson[k]]+val[rson[k]];
    }

    void update(int &k,int l,int r,int x,int v){
        if(!k) k=++tot;
        if(l==r){
            val[k]+=v;
            return;
        }
        int mid=l+(r-l)/2;
        if(x<=mid) update(lson[k],l,mid,x,v);
        else update(rson[k],mid+1,r,x,v);
        pushup(k);
    }

    long long getCnt(int k,int l,int r,int x,int y){
        if(!k) return 0;
        if(x<=l&&r<=y) return val[k];
        int mid=l+(r-l)/2;
        long long res=0;
        if(x<=mid) res+=getCnt(lson[k],l,mid,x,y);
        if(mid<y) res+=getCnt(rson[k],mid+1,r,x,y);
        return res;
    }

    int getKth(int k,int l,int r,int v){
        if(!k) return -1;
        if(l==r) return l;
        int mid=l+(r-l)/2;
        if(v<=val[lson[k]]) return getKth(lson[k],l,mid,v);
        else return getKth(rson[k],mid+1,r,v-val[lson[k]]);
    }

    void split(int k,int &kNew,int l,int r,int x,int y){
        if(!k) return;
        if(!kNew) kNew=++tot;
        if(x<=l&&r<=y){
            val[kNew]=val[k];
            lson[kNew]=lson[k];
            rson[kNew]=rson[k];
            val[k]=rson[k]=lson[k]=0;
            return;
        }
        int mid=l+(r-l)/2;
        if(x<=mid) split(lson[k],lson[kNew],l,mid,x,y);
        if(mid<y) split(rson[k],rson[kNew],mid+1,r,x,y);
        pushup(k);
        pushup(kNew);
    }

    int merge(int k1,int k2,int l,int r){
        if(!k1||!k2) return k1+k2;
        if(l==r){
            val[k1]+=val[k2];
            return k1;
        }
        int mid=l+(r-l)/2;
        lson[k1]=merge(lson[k1],lson[k2],l,mid);
        rson[k1]=merge(rson[k1],rson[k2],mid+1,r);
        pushup(k1);
        return k1;
    }

    void printTree(int k,int l,int r,int layer){
        if(!k) return;
        int mid=l+(r-l)/2;
        printTree(lson[k],l,mid,layer+1);
        for(int i=1;i<=layer;i++) printf(" ");
        printf("|");
        for(int i=1;i<=6-layer;i++) printf("_");

        printf("L:%d R:%d %d\n",l,r,val[k]);
        printTree(rson[k],mid+1,r,layer+1);
        return;
    }

}tr;

int main(){
    // freopen("test.in","r",stdin);
    scanf("%d%d",&n,&m);
    TC=1;
    for(int i=1;i<=n;i++){
        int a;
        scanf("%d",&a);
        if(!a) continue;
        tr.update(root[1],1,m,i,a);
    }
    for(int i=1;i<=m;i++){
        int op,a,b,c;
        scanf("%d%d%d",&op,&a,&b);
        if(op==0||op==2||op==3) scanf("%d",&c);
        if(op==0) tr.split(root[a],root[++TC],1,m,b,c);
        else if(op==1){
            // puts("####____BEFORE");
            // tr.printTree(root[a],1,m,0);
            root[a]=tr.merge(root[a],root[b],1,m),root[b]=0;
            // puts("####____AFTER");
            // tr.printTree(root[a],1,m,0);
            // puts("");
        }
        else if(op==2) tr.update(root[a],1,m,c,b);
        else if(op==3) printf("%lld\n",tr.getCnt(root[a],1,m,b,c));
        else{
            
            printf("%d\n",tr.getKth(root[a],1,m,b));
        }
    }
    
    return 0;
}
```
