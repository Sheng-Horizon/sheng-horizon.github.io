# 代码

1. AC版本（数论全家桶）

   ``` cpp
   #include<iostream>
   #include<cstdio>
   #include<algorithm>
   #include<cstring>
   using namespace std;
   const int MOD=999911658;
   const int PM=999911659;
   int n,g;
   int a_[6],m_[6]={0,2,3,4679,35617,0};
   int frac[40000],finv[40000];
   int exgcd(int a,int b,int &x,int &y){
       if(b==0){
           x=1,y=0;
           return a;
       }
       int d=exgcd(b,a%b,x,y);
       int z=x;
       x=y,y=z-(a/b)*y;
       return d;
   }
   int qwqmi(int a,int b,int mod){
       int res=1;
       while(b){
           if(b&1) res=(long long)res*a%mod;
           a=(long long)a*a%mod;
           b>>=1;
       }
       return res;
   }
   int inv(int a,int p){
       int x,y;
       int d=exgcd(a,p,x,y);
       return (x/d%p+p)%p;
   }
   int C(int n,int m,int p){
       if(n<0||n>m) return 0;
       return (long long)frac[m]*inv(frac[n],p)%p*inv(frac[m-n],p)%p;
   }
   int lucas(int n,int m,int p){
       if(n<p&& m<p) return C(n,m,p);
       return (long long)C(n%p,m%p,p)*lucas(n/p,m/p,p)%p;
   }
   int main(){
       scanf("%d%d",&n,&g);
       if(g%PM==0){
           puts("0");
           return 0;
       }
       for(int i=1;i<=4;i++){
           frac[0]=1;
           for(int j=1;j<=m_[i];j++) frac[j]=(long long)frac[j-1]*j%m_[i];
           for(int j=1;j*j<=n;j++){
               if(n%j==0){
                   a_[i]=(a_[i]+lucas(j,n,m_[i]))%m_[i];
                   if(j*j!=n) a_[i]=(a_[i]+lucas(n/j,n,m_[i]))%m_[i];
               }
           }
       }
       int ans=0;
       for(int i=1;i<=4;i++){
           int M=MOD/m_[i];
           ans=((long long)ans+(long long)a_[i]*M%MOD*inv(M,m_[i])%MOD)%MOD;
       }
       printf("%d\n",qwqmi(g,ans,PM));
       return 0;
   }
   ```

2. TLE版本（exlucas）

   ``` cpp
   #include<iostream>
   #include<cstdio>
   #include<cstring>
   #include<algorithm>
   using namespace std;
   typedef long long LL;
   const int MOD=999911658;
   const int PM=999911659;
   const int N=1e5+10;
   int n,g;
   int sum;
   int a_[6],m_[6]={0,2,3,4679,35617,0};
   
   int exgcd(int a,int b,int &x,int &y){
       if(b==0){
           x=1,y=0;
           return a;
       }
       int d=exgcd(b,a%b,x,y);
       int z=x;
       x=y,y=z-(a/b)*y;
       return d;
   }
   
   int inv(int a,int p){
       int x,y;
       int d=exgcd(a,p,x,y);
       return (x/d%p+p)%p;
   }
   
   
   
   int qwqmi(int a,int b,int p){
       int res=1;
       while(b){
           if(b&1) res=(LL)res*a%p;
           a=(LL)a*a%p;
           b>>=1;
       }
       return res;
   }
   
   int F(int n,int p){
       if(n==0) return 1;
       int cycle =1,rem=1;
       for(int i=1;i<=p;i++) if(i%p) cycle=(LL)cycle*i%p;
       cycle=qwqmi(cycle,n/p,p);
       for(int i=(n/p)*p;i<=n;i++){
           if(i%p) rem=(LL)rem*(i%p)%p;
       }
       return (LL)F(n/p,p)*cycle%p*rem%p;
   }
   
   int G(int n,int p){
       if(n<p) return 0;
       return G(n/p,p)+(n/p);
   }
   
   int C(int n,int m,int p){
       int f1=F(m,p);
       int f2=inv(F(n,p),p);
       int f3=inv(F(m-n,p),p);
       int gp=qwqmi(p,G(m,p)-G(n,p)-G(m-n,p),p);
       return (LL)f1*f2%p*(LL)f3%p*(LL)gp%p;
   }
   
   
   int exlucas(int n,int m){
       for(int i=1;i<=4;i++){
           a_[i]=C(n,m,m_[i]);
       }
       int ans=0;
       for(int i=1;i<=4;i++){
           int M=MOD/m_[i];
           ans=((LL)ans+(LL)a_[i]*M%MOD*inv(M,m_[i])%MOD)%MOD;
       }
       return ans;
   }
   
   int main(){
       // freopen("test.in","r",stdin);
       scanf("%d%d",&n,&g);
       if(g%PM==0){
           printf("0\n");
           return 0;
       }
       for(int i=1;i*i<=n;i++){
           if(n%i==0){
               sum=(sum+exlucas(i,n))%MOD;
               if(i*i!=n) sum=(sum+exlucas(n/i,n))%MOD;
           }
       }
       printf("%d\n",qwqmi(g,sum,PM));
       return 0;
   }
	```
	

那么为什么 $exlucas$ 会超时呢？

有一个重要原因是对比两份代码，一份的 $CRT$ 在循环外，一份在循环内，于是 $exlucas$ 复杂度就炸掉了

但是数论全家桶版本中的 $lucas$ 也要做小小的优化，要预处理一个 $frac[i]=x,其中 x \equiv i! \pmod{p}$ 的阶乘数组。**注意**此处还不能像平时一样预处理一个 $inv$ 逆元数组，因为在 $inv[p]$ 处，处理出来直接等于 $0$ 了，处理出来的东西后面都是不对的。
