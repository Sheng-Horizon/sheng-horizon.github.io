# 题目描述

求

$${\mathrm{C}}_n^m \bmod{p}$$

其中 $\mathrm{C}$ 为组合数。

## 输入格式

一行三个整数 $n,m,p$ ，含义由题所述。

## 输出格式

一行一个整数，表示答案。

## 提示

对于 $100 \%$ 的数据，$1 \le m \le n \le {10}^{18}$，$2 \le p \le {10}^6$，**不保证** $p$ 是质数。

# 关键公式

$$
\begin{equation} 
	\begin{aligned}
	n! &= 1 \times 2\times3\times\dots\times n\\
	&= P^{\lfloor\frac{n}{p}\rfloor}(\lfloor\frac{n}{p}\rfloor)!(\prod_{i=1,i\not\equiv{0}(\mod P)}^{P^k}i)^{\lfloor\frac{b}{P^k}\rfloor}(\prod_{i=P^k,i\not\equiv0(\mod p)}i)
	\end{aligned}
\end {equation}
$$

# 代码
```cpp
#include<iostream>
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<climits>
using namespace std;
typedef long long LL;
const int N=1e4+10;
LL n,m,p;
LL a_[N],m_[N];
int tot;

LL qwqmi(LL a,LL b,LL mod){
    LL res=1;
    while(b){
        if(b&1) res=res*a%mod;
        a=a*a%mod;
        b>>=1;
    }
    return res;
}

LL F(LL n,LL p,LL sp){
    if(n==0) return 1LL;
    LL cycle=1,rem=1;
    for(LL i=1;i<=p;i++){
        if(i%sp) cycle=cycle*i%p;
    }
    cycle=qwqmi(cycle,n/p,p);
    for(LL i=p*(n/p);i<=n;i++){
        if(i%sp) rem=rem*(i%p)%p;
    }
    return F(n/sp,p,sp)*cycle%p*rem%p;
}

LL G(LL n,LL sp){
    if(n<sp) return 0;
    return n/sp+G(n/sp,sp); 
}

LL exgcd(LL a,LL b,LL &x,LL &y){
    if(b==0){
        x=1,y=0;
        return a;
    }
    LL d=exgcd(b,a%b,x,y);
    LL z=x;
    x=y,y=z-(a/b)*y;
    return d;
}

LL inv(LL a,LL p){
    LL x,y;
    LL d=exgcd(a,p,x,y);
    return (x%p+p)%p;
}

LL C(LL n,LL m,LL p,LL sp){//n,m,p^k,p单体
    LL f1=F(m,p,sp);
    LL f2=inv(F(n,p,sp),p),f3=inv(F(m-n,p,sp),p);
    LL gp=qwqmi(sp,G(m,sp)-G(n,sp)-G(m-n,sp),p);
    return f1*f2%p*f3%p*gp%p;
}

LL exlucas(LL n,LL m,LL p){
    LL pp=p;
    for(LL i=2;i*i<=pp;i++){
        if(pp%i==0){
            tot++;
            m_[tot]=1;
            while(pp%i==0){
                pp/=i;
                m_[tot]*=i;
            }
            a_[tot]=C(n,m,m_[tot],i);
        }
    }
    LL ans=0;
    if(pp!=1) m_[++tot]=pp,a_[tot]=C(n,m,pp,pp);
    for(LL i=1;i<=tot;i++){
        LL M=p/m_[i];
        ans=(ans+a_[i]*M%p*inv(M,m_[i])%p)%p;
    }
    return ans;
}

int main(){
    // freopen("test.in","r",stdin);
    scanf("%lld%lld%lld",&n,&m,&p);
    printf("%lld\n",exlucas(m,n,p));
    return 0;
}
```
